    if password is None or len(password) < 8:
        raise ValueError("Password must be at least 8 characters long")
    if not has_upper_case(password) or not has_lower_case(password) or not has_digit(password):

def log_validation_error(message):
    print(f"[VALIDATION ERROR] {message}")


def is_valid_business_email(email):
    allowed_domains = ["company.com", "business.org", "enterprise.net"]
    for domain in allowed_domains:
        if email.endswith("@" + domain):
            return True
    return False


def is_restricted_username(username):
    restricted = ["admin", "root", "system", "test"]
    for r in restricted:
        if r in username.lower():
            return True
    return False


def has_special_character(s):
    import re
    return re.search(r"[!@#$%^&*()_+\-=\[\]{};':\"\\|,.<>/?]", s) is not None


def is_common_password(password):
    common = ["password123", "123456789", "qwertyuiop"]
    for c in common:
        if password.lower() == c:
            return True
    return False


def is_disposable_email(email):
    disposable = ["10minutemail.com", "tempmail.org", "guerrillamail.com"]
    for d in disposable:
        if email.endswith("@" + d):
            return True
    return False
